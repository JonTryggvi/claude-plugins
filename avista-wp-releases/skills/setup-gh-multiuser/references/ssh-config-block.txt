# --- Avista GitHub multi-account SSH config ---------------------------------
# Append to ~/.ssh/config. Substitute the IdentityFile paths with the actual
# keys generated for this machine. The `github.com-avista` alias is what the
# Avista org remotes use (git@github.com-avista:Avista/<repo>.git) and what
# set_gh_user() keys off to pick the org account.

# Avista org account — used by github.com-avista:Avista/* remotes
Host github.com-avista
  HostName github.com
  User git
  IdentityFile ~/.ssh/id_ed25519_avista
  IdentitiesOnly yes

# Personal account — default github.com host
Host github.com
  HostName github.com
  User git
  IdentityFile ~/.ssh/id_ed25519
  IdentitiesOnly yes
